cmake_minimum_required(VERSION 3.16)
project(sl2f13_exact_verification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(GMP_INCLUDE_DIR gmpxx.h REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)
find_library(GMP_LIBRARY gmp REQUIRED)
foreach(target norm13 norm28)
  add_executable(${target} ${target}.cpp)
  target_include_directories(${target} PRIVATE ${GMP_INCLUDE_DIR})
  target_link_libraries(${target} PRIVATE ${GMPXX_LIBRARY} ${GMP_LIBRARY})
  target_compile_options(${target} PRIVATE -O2)
endforeach()
